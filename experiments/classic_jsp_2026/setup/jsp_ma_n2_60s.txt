# Memetic algorithm (GA + tabu search on N2) on the classic JSP.
#
# Derived from jsp_abc_n2_60s.txt with algorithm = MEMETIC and nothing else
# changed, so that the memetic and the ABC are compared like for like: same
# operators, same local search, same budget. Written to check that the in-place
# neighbour evaluation of N2 behaves under the other algorithm that uses it.


algorithm = MEMETIC
seed = 1
runs = 10

# PROBLEM
problem.hasbounds = no

# OBJECTIVE
objective = ijsp.makespan
evaluation.lamarckism = yes

# POST EXECUTION
# The interval robustness analysers are gone with the intervals; a run emits a
# verifiable schedule instead, which is what verify_certificate.py checks.
postexecution.analyzer = jsp.certificate

# CODIFICATION
codification = ijsp.job-order
decodification = ijsp.job-order
decode.sgs = insertion

# CREATION
creation = ijsp.random
creation.sgs = insertion

# CROSSOVER
crossover = ijsp.jox
crossover.probability = 0.9

# MUTATION
mutation = insertion
mutation.probability = 0.10

# SELECTION / REPLACEMENT
selection = shuffle
replacement = parents
replacement.allow-repeated = no

# STOPPING CRITERION
noimprovement = 100000
timelimit = 60

# POPULATION
population.size = 250
maxnumtrials = 24
elite.selection = 6
elite.size = 50

# STATISTICS
statistics.1.metric = best
statistics.1.value = ijsp.makespan
statistics.2.metric = average
statistics.2.value = ijsp.makespan

# LOCAL SEARCH
localsearch = tabu-search
localsearch.estimation-guide = no
localsearch.filter = yes
localsearch.max-time = 2
localsearch.neighbourhood = ijsp.makespan.n2
neighbourhood.estimator = heads&tails
localsearch.frequency = period
localsearch.period = 1
localsearch.target = 1.0
localsearch.lamarckism = yes
localsearch.bad-iterations = 20
localsearch.tabu-size.min = 3
