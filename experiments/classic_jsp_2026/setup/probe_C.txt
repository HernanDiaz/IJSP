# Classic JSP (crisp) — ABCPSO + tabu search on N2.
#
# A classic JSP instance is an IJSP instance whose processing times are all
# degenerate intervals [p, p]. With a == b every Interval ranking method
# collapses to the order on the reals and the component-wise maximum is the
# ordinary maximum, so this configuration optimises the classic makespan
# exactly. LEX2 is kept because it is the comparison the irace tuning of the
# COR-2026 study selected; on crisp data it is equivalent to EV.
#
# Hyperparameters are those of experiments/cor_tabu_2026/setup/setup_N2_tuned.txt
# except for the stopping rule: the tuned setup stops after 20 generations
# without improvement, which on crisp Taillard instances triggers while the run
# is still improving, so the search is given the full time budget instead.

# GENERIC
algorithm = ABCPSO
seed = 1
runs = 3

# PROBLEM
problem.hasbounds = no

# OBJECTIVE
objective = ijsp.makespan
evaluation.interval.comparison = LEX2
evaluation.interval.maximum = Component
evaluation.lamarckism = yes

# POST EXECUTION
# The robustness analysis samples durations inside each interval, which is a
# no-op on degenerate intervals; emit a verifiable schedule instead.
postexecution.analyzer = jsp.certificate

# CODIFICATION
codification = ijsp.job-order
decodification = ijsp.job-order
decode.sgs = insertion
sgs.interval.comparison = LEX2

# CREATION
creation = ijsp.random
creation.sgs = insertion

# CROSSOVER
crossover = ijsp.jox
crossover.probability = 0.9

# MUTATION
mutation = insertion
mutation.probability = 0.10

# SELECTION / REPLACEMENT
selection = shuffle
replacement = parents
replacement.allow-repeated = no

# STOPPING CRITERION
noimprovement = 100000
timelimit = 60

# POPULATION
population.size = 250
maxnumtrials = 24
elite.selection = 6
elite.size = 50

# STATISTICS
statistics.1.metric = best
statistics.1.value = ijsp.makespan
statistics.2.metric = average
statistics.2.value = ijsp.makespan

# LOCAL SEARCH
localsearch = tabu-search
localsearch.estimation-guide = no
localsearch.filter = yes
localsearch.max-time = 5
localsearch.neighbourhood = ijsp.makespan.n2
neighbourhood.estimator = heads&tails
localsearch.frequency = period
localsearch.period = 1
localsearch.target = 1.0
localsearch.lamarckism = yes
localsearch.bad-iterations = 500
localsearch.tabu-size.min = 3
localsearch.tabu-size.max = 12
