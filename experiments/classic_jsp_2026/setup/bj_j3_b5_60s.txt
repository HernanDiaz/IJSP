# Classic JSP (crisp) — ABCPSO + tabu search on N2.
#
# Processing times are crisp integers. The interval machinery this branch grew
# out of is gone: there is one order on the times and std::max is the maximum,
# so the ranking strategy this file used to select (LEX2, from the irace tuning
# of the COR-2026 study, equivalent to EV on crisp data) no longer names a
# choice and is no longer set.
#
# Hyperparameters are those of experiments/cor_tabu_2026/setup/setup_N2_tuned.txt
# except for the stopping rule: the tuned setup stops after 20 generations
# without improvement, which on crisp Taillard instances triggers while the run
# is still improving, so the search is given the full time budget instead.

# GENERIC
algorithm = ABCPSO
seed = 1
runs = 10

# PROBLEM
problem.hasbounds = no

# OBJECTIVE
objective = jsp.makespan
evaluation.lamarckism = yes

# POST EXECUTION
# The interval robustness analysers are gone with the intervals; a run emits a
# verifiable schedule instead, which is what verify_certificate.py checks.
postexecution.analyzer = jsp.certificate

# CODIFICATION
codification = jsp.job-order
decodification = jsp.job-order
decode.sgs = insertion

# CREATION
creation = jsp.random
creation.sgs = insertion

# CROSSOVER
crossover = jsp.jox
crossover.probability = 0.9

# MUTATION
mutation = insertion
mutation.probability = 0.10

# SELECTION / REPLACEMENT
selection = shuffle
replacement = parents
replacement.allow-repeated = no

# STOPPING CRITERION
noimprovement = 100000
timelimit = 60

# POPULATION
population.size = 250
maxnumtrials = 24
elite.selection = 6
elite.size = 50

# STATISTICS
statistics.1.metric = best
statistics.1.value = jsp.makespan
statistics.2.metric = average
statistics.2.value = jsp.makespan

# LOCAL SEARCH
localsearch = tabu-backjump
localsearch.estimation-guide = no
localsearch.filter = yes
localsearch.max-time = 2
localsearch.neighbourhood = jsp.makespan.n2
neighbourhood.estimator = heads&tails
localsearch.frequency = period
localsearch.period = 1
localsearch.target = 1.0
localsearch.lamarckism = yes
localsearch.bad-iterations = 5
localsearch.tabu-size.min = 3

# Back-jump: at most 3 jumps per call, 5 bad iterations per pass, so the
# total work per call stays comparable to the 20 of the plain tabu baseline.
localsearch.backjump.size = 5
localsearch.backjump.max-jumps = 3
