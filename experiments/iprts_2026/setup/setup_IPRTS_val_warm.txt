# IPRTS v1 warm-start: pool seeded from the paper TS-N2 solutions on the 12 selected classical instances.
# Same configuration as setup_IPRTS_base.txt with a reduced budget:
# 5 runs x 60 s (>= the median runtime of tuned TS-N2 on every classical
# instance, 6.5-40 s per the paper's per_instance_stats.csv).

# GENERIC
algorithm = IPRTS
seed = 1
runs = 5

# PROBLEM
problem.hasbounds = no

# OBJECTIVE
objective = ijsp.makespan
evaluation.interval.comparison = LEX2
evaluation.interval.maximum = Component
evaluation.lamarckism = yes

# CODIFICATION
codification = ijsp.job-order
decodification = ijsp.job-order
decode.sgs = insertion
sgs.interval.comparison = LEX2

# CREATION (pool seeding)
creation = ijsp.solutions-file
creation.solutions-dir = experiments/iprts_2026/seeds/N2_tuned
creation.randomratio = 0.2
creation.sgs = insertion

# ELITE POOL
pool.size = 10
pool.min-distance = 0.05
pool.restart-patience = 50
pool.seed-retries = 5

# PATH RELINKING
pr.neighbourhood = ijsp.makespan.n1
pr.max-steps = -1

# STOPPING CRITERION (noimprovement counts PR cycles)
noimprovement = 5000
timelimit = 60

# EVOLUTION REPORTING
evolution.unit = time
evolution.span = 1

# STATISTICS
statistics.1.metric = best
statistics.1.value = ijsp.makespan
statistics.2.metric = average
statistics.2.value = ijsp.makespan

# LOCAL SEARCH (irace-tuned TS-N2 values from the COR paper)
localsearch = tabu-search
localsearch.estimation-guide = no
localsearch.filter = no
localsearch.max-time = 2
localsearch.neighbourhood = ijsp.makespan.n2
neighbourhood.estimator = heads&tails
localsearch.lamarckism = yes
localsearch.bad-iterations = 100
localsearch.tabu-size.min = 3
localsearch.tabu-size.max = 8
localsearch.max-rounds = 50
